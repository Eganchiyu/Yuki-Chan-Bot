"""命令行入口：python -m diary_graph {build,serve}"""

from __future__ import annotations

import argparse
import functools
import hashlib
import http.server
import json
import socketserver
import sys
import time
from pathlib import Path

from .config import DEFAULT_DB, DEFAULT_EXTRA_STOP, DEFAULT_OUT, ROOT, BuildOptions
from .extract import KeywordExtractor
from .graph import index_node_docs, stats_markdown
from .graph import analyze, build_graph
from .loader import load_diaries
from .render import render

CACHE_NAME = "keywords_cache.json"


def _log(msg: str) -> None:
    print(msg, file=sys.stderr, flush=True)


def _lexicon_fingerprint(top_k: int, extra_stop=None) -> str:
    """词典/停用词/别名任一改动都应让缓存失效。"""
    h = hashlib.md5(f"topk={top_k}".encode())
    files = [ROOT / "custom_dict.txt", ROOT / "stopwords.txt", ROOT / "aliases.txt"]
    if extra_stop:
        files.append(Path(extra_stop))
    for f in files:
        p = Path(f)
        h.update(p.read_bytes() if p.exists() else b"-")
    return h.hexdigest()[:16]


def _extract_all(diaries, opts: BuildOptions, cache_path: Path, extra_stop=None):
    extractor = KeywordExtractor(ROOT, extra_stop=extra_stop)
    fingerprint = _lexicon_fingerprint(opts.top_k, extra_stop)
    cache: dict[str, dict] = {}
    if opts.use_cache and cache_path.exists():
        try:
            blob = json.loads(cache_path.read_text(encoding="utf-8"))
            if blob.get("fingerprint") == fingerprint:
                cache = blob.get("items", {})
        except (json.JSONDecodeError, AttributeError):
            cache = {}

    keywords: list[list[str]] = []
    hits = 0
    t0 = time.time()
    for i, diary in enumerate(diaries, 1):
        digest = hashlib.md5(diary.text.encode("utf-8")).hexdigest()[:12]
        entry = cache.get(diary.id)
        if entry and entry.get("h") == digest:
            keywords.append(entry["kw"])
            hits += 1
        else:
            kw = extractor.extract(diary.text, top_k=opts.top_k)
            keywords.append(kw)
            cache[diary.id] = {"h": digest, "kw": kw}
        if i % 500 == 0:
            _log(f"  关键词抽取 {i}/{len(diaries)}（缓存命中 {hits}）…")
    _log(f"关键词抽取完成，用时 {time.time() - t0:.1f}s，缓存命中 {hits}/{len(diaries)}")

    if opts.use_cache:
        cache_path.parent.mkdir(parents=True, exist_ok=True)
        cache_path.write_text(
            json.dumps({"fingerprint": fingerprint, "items": cache}, ensure_ascii=False),
            encoding="utf-8")
    return keywords


def cmd_build(args: argparse.Namespace) -> int:
    opts = BuildOptions(
        db=Path(args.db), out_dir=Path(args.out), collection=args.collection,
        limit=args.limit, top_k=args.top_k, halflife_days=args.halflife,
        min_weight=args.min_weight, max_nodes=args.max_nodes,
        max_docs_per_node=args.max_docs, snippet_len=args.snippet_len,
        use_cache=not args.no_cache,
    )
    _log(f"读取记忆库: {opts.db}（collection={opts.collection}）")
    diaries = load_diaries(opts.db, opts.collection, limit=opts.limit or None)
    if not diaries:
        _log("没有读到任何日记，检查 --db / --collection")
        return 1
    _log(f"日记 {len(diaries)} 篇，时间范围 {diaries[0].date} ~ {diaries[-1].date}")

    keywords = _extract_all(diaries, opts, opts.out_dir / CACHE_NAME,
                            extra_stop=args.extra_stop)
    entries = [(kw, d.timestamp) for d, kw in zip(diaries, keywords)]

    _log(f"建图: 半衰期 {opts.halflife_days:g} 天，剪枝阈值 {opts.min_weight}，最多保留 {opts.max_nodes or '不限'} 节点")
    G = build_graph(entries, halflife_days=opts.halflife_days,
                    min_weight=opts.min_weight, max_nodes=opts.max_nodes)
    if G.number_of_nodes() == 0:
        _log("剪枝后图为空，请降低 --min-weight 或提高 --max-nodes")
        return 1

    _log(f"图规模 {G.number_of_nodes()} 节点 / {G.number_of_edges()} 边，计算簇与桥接…")
    bundle = analyze(G, resolution=args.resolution)
    bundle.node_docs = index_node_docs(
        diaries, keywords, set(G.nodes()), max_per_node=opts.max_docs_per_node,
        snippet_len=opts.snippet_len)

    html_path = render(bundle, opts.out_dir / "mind_map.html",
                       title=f"Yuki 日记图谱（{len(diaries)} 篇）")
    md_path = opts.out_dir / "graph_stats.md"
    md_path.write_text(stats_markdown(bundle.stats), encoding="utf-8")
    (opts.out_dir / "graph_stats.json").write_text(
        json.dumps(bundle.stats, ensure_ascii=False, indent=2), encoding="utf-8")

    _log(f"完成:\n  {html_path}\n  {md_path}")
    print(f"{bundle.stats['nodes']}\t{bundle.stats['edges']}\t{bundle.stats['communities']}")
    return 0


def cmd_serve(args: argparse.Namespace) -> int:
    root = Path(args.dir).resolve()
    if not root.exists():
        _log(f"目录不存在: {root}，先跑 build")
        return 1
    handler = functools.partial(http.server.SimpleHTTPRequestHandler, directory=str(root))
    socketserver.TCPServer.allow_reuse_address = True
    with socketserver.TCPServer(("127.0.0.1", args.port), handler) as httpd:
        _log(f"服务中: http://127.0.0.1:{args.port}/mind_map.html  （Ctrl-C 结束）")
        try:
            httpd.serve_forever()
        except KeyboardInterrupt:
            _log("已停止")
    return 0


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        prog="python -m diary_graph",
        description="Yuki 日记共现图谱：建图 + 物理仿真可视化（离线单文件 HTML）")
    sub = p.add_subparsers(dest="cmd", required=True)

    b = sub.add_parser("build", help="抽取关键词、建图、输出 HTML 与统计")
    b.add_argument("--db", default=str(DEFAULT_DB), help="chroma.sqlite3 路径")
    b.add_argument("--out", default=str(DEFAULT_OUT), help="输出目录")
    b.add_argument("--collection", default="diaries")
    b.add_argument("--limit", type=int, default=0, help="只用最新 N 篇（0=全部）")
    b.add_argument("--top-k", type=int, default=8, help="每篇关键词数")
    b.add_argument("--halflife", type=float, default=365.0,
                   help="时间衰减半衰期（天），<=0 关闭")
    b.add_argument("--min-weight", type=float, default=1.0, help="边权重剪枝阈值")
    b.add_argument("--max-nodes", type=int, default=900,
                   help="保留加权度最高的 N 个概念（0=不限）")
    b.add_argument("--max-docs", type=int, default=8, help="每个概念最多挂几篇日记")
    b.add_argument("--snippet-len", type=int, default=260, help="日记片段长度")
    b.add_argument("--extra-stop", default=str(DEFAULT_EXTRA_STOP),
                   help="额外屏蔽词文件（默认复用主项目 blacklist.txt）")
    b.add_argument("--resolution", type=float, default=1.0,
                   help="Louvain 分辨率，越大簇越细")
    b.add_argument("--no-cache", action="store_true", help="忽略关键词缓存")
    b.set_defaults(func=cmd_build)

    s = sub.add_parser("serve", help="本地 http 服务打开生成的图谱")
    s.add_argument("--port", type=int, default=8779)
    s.add_argument("--dir", default=str(DEFAULT_OUT))
    s.set_defaults(func=cmd_serve)
    return p


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    return args.func(args)


if __name__ == "__main__":
    raise SystemExit(main())

"""本地关键词抽取：jieba 分词 + 词性过滤 + TF-IDF Top-K。"""

from __future__ import annotations

import re
from pathlib import Path

import jieba
import jieba.analyse
import jieba.posseg as pseg  # noqa: F401  (加载 posseg 资源)

# 只保留有概念价值的词性
ALLOW_POS = ("n", "nr", "ns", "nt", "nz", "vn", "v", "eng")

_PURE_DIGIT = re.compile(r"^[\d.\-_%]+$")
_URLISH = re.compile(r"(https?://|www\.)")
# 日记正文的时间前缀不属于内容
_DIARY_PREFIX = re.compile(r"^\s*【日记\([^)）]*\)】[：:]?\s*")


class KeywordExtractor:
    def __init__(self, base_dir: str | Path, min_len: int = 2,
                 allow_pos: tuple[str, ...] = ALLOW_POS,
                 extra_stop: str | Path | None = None):
        base = Path(base_dir)
        self.min_len = min_len
        self.allow_pos = allow_pos
        self.aliases = self._load_aliases(base / "aliases.txt")
        self.stopwords = self._load_stopwords(base / "stopwords.txt")
        if extra_stop:
            self.stopwords |= self._load_stopwords(Path(extra_stop))

        custom = base / "custom_dict.txt"
        if custom.exists():
            jieba.load_userdict(str(custom))
        if self.stopwords:
            jieba.analyse.set_stop_words(str(base / "stopwords.txt"))
        jieba.initialize()

        # 别名替换按长度降序，避免短别名先命中
        self._alias_patterns = [
            (re.compile(re.escape(alias), re.IGNORECASE), canon)
            for canon, alias in sorted(
                ((c, a) for c, al in self.aliases.items() for a in al),
                key=lambda x: len(x[1]), reverse=True)
        ]

    @staticmethod
    def _load_aliases(path: Path) -> dict[str, list[str]]:
        out: dict[str, list[str]] = {}
        if not path.exists():
            return out
        for line in path.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line or line.startswith("#") or "=" not in line:
                continue
            canon, al = line.split("=", 1)
            out[canon.strip()] = [a.strip() for a in al.split("|") if a.strip()]
        return out

    @staticmethod
    def _load_stopwords(path: Path) -> set[str]:
        if not path.exists():
            return set()
        return {
            line.strip().lower()
            for line in path.read_text(encoding="utf-8").splitlines()
            if line.strip() and not line.startswith("#")
        }

    def normalize(self, text: str) -> str:
        text = _DIARY_PREFIX.sub("", text)
        for pat, canon in self._alias_patterns:
            text = pat.sub(canon, text)
        return text

    def _keep(self, word: str) -> bool:
        w = word.strip()
        if len(w) < self.min_len or _PURE_DIGIT.match(w) or _URLISH.search(w):
            return False
        return w.lower() not in self.stopwords

    def extract(self, text: str, top_k: int = 8) -> list[str]:
        text = self.normalize(text)
        if not text:
            return []
        tags = jieba.analyse.extract_tags(
            text, topK=max(top_k * 3, top_k + 4), allowPOS=self.allow_pos,
        )
        picked: list[str] = []
        for w in tags:
            if self._keep(w) and w not in picked:
                picked.append(w)
            if len(picked) >= top_k:
                break
        return picked

"""共现建图 + 时间衰减 + 剪枝 + 图分析指标。"""

from __future__ import annotations

import itertools
import math
import re
from collections import Counter, defaultdict
from dataclasses import dataclass, field

import networkx as nx

from .loader import Diary

_PREFIX = re.compile(r"^\s*【日记\([^)）]*\)】[：:]?\s*")


@dataclass
class GraphBundle:
    graph: nx.Graph
    communities: dict[str, int] = field(default_factory=dict)
    metrics: dict[str, dict] = field(default_factory=dict)
    node_docs: dict[str, list[dict]] = field(default_factory=dict)
    stats: dict = field(default_factory=dict)


def _decay(days_ago: float, halflife_days: float) -> float:
    if halflife_days <= 0:
        return 1.0
    return math.pow(2.0, -max(days_ago, 0.0) / halflife_days)


def build_graph(
    entries: list[tuple[list[str], float]],
    *,
    now: float | None = None,
    halflife_days: float = 365.0,
    min_weight: float = 0.0,
    max_nodes: int = 0,
) -> nx.Graph:
    """entries = [(关键词列表, 日记时间戳)]，返回剪枝后的无向加权图。"""
    if now is None:
        now = max((ts for _, ts in entries), default=0.0)
    now = max(now, max((ts for _, ts in entries), default=0.0))

    G = nx.Graph()
    for kws, ts in entries:
        uniq = list(dict.fromkeys(k for k in kws if k))
        if len(uniq) < 2:
            G.add_nodes_from(uniq)
            continue
        w = _decay((now - ts) / 86400.0, halflife_days)
        for u, v in itertools.combinations(uniq, 2):
            if G.has_edge(u, v):
                G[u][v]["weight"] += w
            else:
                G.add_edge(u, v, weight=w)

    if min_weight > 0:
        G.remove_edges_from([(u, v) for u, v, d in G.edges(data=True)
                             if d["weight"] < min_weight])
    G.remove_nodes_from([n for n, d in G.degree() if d == 0])
    if G.number_of_edges() == 0:
        return G

    if max_nodes and G.number_of_nodes() > max_nodes:
        keep = sorted(G.degree(weight="weight"), key=lambda x: x[1], reverse=True)
        G = G.subgraph([n for n, _ in keep[:max_nodes]]).copy()
        G.remove_nodes_from([n for n, d in G.degree() if d == 0])
    return G


def analyze(G: nx.Graph, seed: int = 42, betweenness_samples: int = 400,
            resolution: float = 1.0) -> GraphBundle:
    """社区划分 + 度/加权度/桥接度/介数中心性。"""
    stats: dict = {}
    if G.number_of_nodes() == 0:
        return GraphBundle(graph=G, stats={"nodes": 0, "edges": 0})

    communities = nx.community.louvain_communities(
        G, weight="weight", seed=seed, resolution=resolution)
    comm_of: dict[str, int] = {}
    for cid, members in enumerate(sorted(communities, key=len, reverse=True)):
        for n in members:
            comm_of[n] = cid

    n_nodes = G.number_of_nodes()
    if n_nodes > 2000:
        k = min(betweenness_samples, n_nodes)
        betweenness = nx.betweenness_centrality(G, k=k, seed=seed)
    else:
        betweenness = nx.betweenness_centrality(G)

    degree = dict(G.degree())
    wdegree = dict(G.degree(weight="weight"))
    clustering = nx.clustering(G, weight=None)

    metrics: dict[str, dict] = {}
    for n in G.nodes():
        neighbor_comms = {comm_of.get(nb, -1) for nb in G.neighbors(n)}
        metrics[n] = {
            "degree": degree[n],
            "weighted_degree": round(wdegree[n], 3),
            "betweenness": round(betweenness.get(n, 0.0), 6),
            # 桥接：邻居里出现了多少个不同簇，越多越像"跨领域十字路口"
            "bridging": len(neighbor_comms - {comm_of.get(n, -1)}),
            "clustering": round(clustering.get(n, 0.0), 3),
            "community": comm_of.get(n, -1),
        }

    stats.update({
        "nodes": n_nodes,
        "edges": G.number_of_edges(),
        "components": nx.number_connected_components(G),
        "largest_component": max(len(c) for c in nx.connected_components(G)),
        "communities": len(communities),
        "avg_degree": round(2 * G.number_of_edges() / n_nodes, 2),
        "density": round(nx.density(G), 6),
        "top_degree": sorted(wdegree.items(), key=lambda x: x[1], reverse=True)[:25],
        "top_betweenness": sorted(
            ((n, m["betweenness"]) for n, m in metrics.items()),
            key=lambda x: x[1], reverse=True)[:25],
        "top_bridging": sorted(
            ((n, m["bridging"], m["degree"]) for n, m in metrics.items()),
            key=lambda x: (x[1], x[2]), reverse=True)[:25],
        "community_sizes": Counter(comm_of.values()).most_common(20),
    })
    return GraphBundle(graph=G, communities=comm_of, metrics=metrics, stats=stats)


def index_node_docs(diaries: list[Diary], keywords: list[list[str]],
                    keep_nodes: set[str], max_per_node: int = 12,
                    snippet_len: int = 260) -> dict[str, list[dict]]:
    """节点 -> 相关日记片段（供前端点节点查看内容）。"""
    node_docs: dict[str, list[dict]] = defaultdict(list)
    for diary, kws in zip(diaries, keywords):
        for k in kws:
            if k not in keep_nodes:
                continue
            bucket = node_docs[k]
            if len(bucket) >= max_per_node:
                continue
            text = _PREFIX.sub("", diary.text)
            bucket.append({
                "id": diary.id,
                "date": diary.date,
                "text": text if len(text) <= snippet_len else text[:snippet_len] + "…",
            })
    return dict(node_docs)


def stats_markdown(stats: dict) -> str:
    lines = [
        "# 日记图谱统计",
        "",
        f"- 节点数: {stats.get('nodes', 0)}",
        f"- 边数: {stats.get('edges', 0)}",
        f"- 簇数(Louvain): {stats.get('communities', 0)}",
        f"- 连通分量: {stats.get('components', 0)}（最大 {stats.get('largest_component', 0)}）",
        f"- 平均度: {stats.get('avg_degree', 0)}　密度: {stats.get('density', 0)}",
        "",
        "## 核心话题（加权度 Top）",
        "",
        *[f"- {n}  `{w}`" for n, w in stats.get("top_degree", [])],
        "",
        "## 桥梁话题（跨簇邻居数 Top）",
        "",
        *[f"- {n}  跨 {b} 簇 / 度 {d}" for n, b, d in stats.get("top_bridging", [])],
        "",
        "## 介数中心性 Top",
        "",
        *[f"- {n}  `{b}`" for n, b in stats.get("top_betweenness", [])],
        "",
        "## 簇规模 Top",
        "",
        *[f"- 簇 {cid}: {size}" for cid, size in stats.get("community_sizes", [])],
    ]
    return "\n".join(lines) + "\n"

"""从 Yuki 的 ChromaDB 只读加载日记。

不依赖 chromadb 包：直接读 chroma.sqlite3 的 embeddings / embedding_metadata /
segments / collections 表，速度快且不触发向量模型加载。
"""

from __future__ import annotations

import datetime as dt
import re
import sqlite3
from dataclasses import dataclass
from pathlib import Path

# 日记正文前缀形如 【日记(2026-11-05 12:00)】：
PREFIX_RE = re.compile(r"【日记\((\d{4})-(\d{2})-(\d{2})(?:\s+(\d{2}):(\d{2}))?")
PREFIX_LEN = 0


@dataclass
class Diary:
    id: str
    text: str
    timestamp: float
    chat_id: str = ""

    @property
    def when(self) -> dt.datetime:
        return dt.datetime.fromtimestamp(self.timestamp)

    @property
    def date(self) -> str:
        return self.when.strftime("%Y-%m-%d %H:%M")


_QUERY = """
SELECT e.id, m.key, m.string_value, m.float_value, m.int_value
FROM embeddings e
JOIN segments s ON s.id = e.segment_id
JOIN collections c ON c.id = s.collection
JOIN embedding_metadata m ON m.id = e.id
WHERE c.name = ? AND m.key IN ('chroma:document', 'timestamp', 'chat_id')
"""


def _coerce_ts(prefix: re.Match | None, raw: float | None) -> float:
    """优先用正文前缀里的时间，其次用 metadata timestamp。"""
    if prefix:
        y, mo, d, hh, mm = prefix.groups()
        try:
            return dt.datetime(
                int(y), int(mo), int(d), int(hh or 0), int(mm or 0)
            ).timestamp()
        except ValueError:
            pass
    return float(raw or 0.0)


def load_diaries(db_path: str | Path, collection: str = "diaries",
                 limit: int | None = None) -> list[Diary]:
    db_path = Path(db_path).expanduser().resolve()
    if not db_path.exists():
        raise FileNotFoundError(f"找不到记忆库文件: {db_path}")

    conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    try:
        rows = conn.execute(_QUERY, (collection,)).fetchall()
    finally:
        conn.close()

    docs: dict[str, dict[str, object]] = {}
    for eid, key, sval, fval, ival in rows:
        rec = docs.setdefault(str(eid), {})
        if key == "chroma:document":
            rec["text"] = sval or ""
        elif key == "timestamp":
            rec["ts"] = fval if fval is not None else ival
        elif key == "chat_id":
            rec["chat_id"] = sval or ""

    diaries: list[Diary] = []
    for eid, rec in docs.items():
        text = str(rec.get("text") or "").strip()
        if not text:
            continue
        m = PREFIX_RE.search(text[:40])
        diaries.append(Diary(
            id=eid,
            text=text,
            timestamp=_coerce_ts(m, rec.get("ts")),  # type: ignore[arg-type]
            chat_id=str(rec.get("chat_id") or ""),
        ))

    diaries.sort(key=lambda d: d.timestamp)
    if limit:
        diaries = diaries[-limit:]
    return diaries

def _size(degree: int, weighted: float) -> float:
    return round(min(10 + 7 * math.log2(degree + 1) + math.log2(weighted + 1), 70), 2)


def render(bundle: GraphBundle, out_path: str | Path, *,
           title: str = "Yuki 日记图谱", height: str = "100vh") -> Path:
    G: nx.Graph = bundle.graph
    out_path = Path(out_path)
    out_path.parent.mkdir(parents=True, exist_ok=True)

    if G.number_of_nodes() == 0:
        out_path.write_text("<!doctype html><meta charset='utf-8'><h1>图为空</h1>",
                            encoding="utf-8")
        return out_path

    max_w = max((d["weight"] for _, _, d in G.edges(data=True)), default=1.0)

    net = Network(height=height, width="100%", bgcolor="#0e0916",
                  font_color="#ffe9f7", directed=False, cdn_resources="in_line")
    net.toggle_physics(True)

    for node, meta in bundle.metrics.items():
        cid = meta["community"]
        color = PALETTE[cid % len(PALETTE)]
        net.add_node(
            node,
            label=node,
            title=(f"{node}\n度 {meta['degree']} | 加权度 {meta['weighted_degree']}\n"
                   f"簇 {cid} | 跨簇桥接 {meta['bridging']} | 介数 {meta['betweenness']}"),
            size=_size(meta["degree"], meta["weighted_degree"]),
            color={"background": color, "border": "#ffffff33",

            font={"size": 15 if meta["degree"] > 6 else 12},
            group=cid,
        )

        w = d["weight"]
        ratio = w / max_w
        net.add_edge(
            u, v,
            value=round(w, 3),
            width=round(min(0.6 + 1.1 * math.log2(w + 1), 7), 2),

            title=f"共现权重 {w:.2f}",
        )

from pathlib import Path
from typing import Iterator
import codecs
import sys

from pathspec import PathSpec


# =========================
# 配置
# =========================

# 合并后的输出文件名
OUTPUT_NAME = "merged_files.txt"

# 文件路径分隔线
SEPARATOR = "================"

# 是否跳过符号链接，避免意外读取项目目录之外的文件
SKIP_SYMLINKS = True

# 根目录：脚本所在文件夹
ROOT = Path(__file__).resolve().parent

# 输出文件绝对路径
OUTPUT_PATH = (ROOT / OUTPUT_NAME).resolve()


# =========================
# .gitignore 处理
# =========================

def load_gitignore(directory: Path):
    """
    读取当前目录下的 .gitignore。
    返回 GitWildMatch 规则集。
    """
    gitignore = directory / ".gitignore"

    if not gitignore.is_file():
        return None

    try:
        lines = gitignore.read_text(
            encoding="utf-8-sig",
            errors="replace"
        ).splitlines()

        return PathSpec.from_lines("gitwildmatch", lines)

    except Exception as exc:
        print(f"[警告] 无法读取 {gitignore}: {exc}", file=sys.stderr)
        return None



def is_ignored(
    path: Path,
    is_directory: bool,
    ignore_rules: list
) -> bool:
    """
    判断文件或文件夹是否被 .gitignore 忽略。
    """
    ignored = False

    for base_dir, spec in ignore_rules:
        try:
            relative_path = path.relative_to(base_dir).as_posix()
        except ValueError:
            continue

        if is_directory:
            relative_path += "/"

        result = spec.check_file(relative_path)

        # True 代表匹配忽略规则
        # False 代表匹配 ! 反向放行规则
        # None 代表没有匹配
        if result.include is not None:
            ignored = result.include

    return ignored



# =========================
# 递归遍历
# =========================

def iter_files(
    directory: Path,
    inherited_rules: list
) -> Iterator[Path]:
    """
    递归遍历目录，按照各级 .gitignore 规则筛选文件。
    """

    # 继承父目录的忽略规则
    current_rules = list(inherited_rules)

    # 加载当前目录的 .gitignore
    local_spec = load_gitignore(directory)

    if local_spec is not None:
        current_rules.append((directory, local_spec))

    try:
        entries = sorted(
            directory.iterdir(),
            key=lambda p: (p.name.casefold(), p.name)
        )
    except PermissionError:
        print(f"[跳过] 无权访问目录：{directory}", file=sys.stderr)
        return
    except OSError as exc:
        print(f"[跳过] 无法访问目录 {directory}: {exc}", file=sys.stderr)
        return

    for entry in entries:

        # 避免读取 .git 内部元数据
        if entry.name == ".git":
            continue

        # 不跟随符号链接
        if SKIP_SYMLINKS and entry.is_symlink():
            continue

        try:
            if entry.is_dir():

                # 被忽略的目录不再向下遍历
                if is_ignored(entry, True, current_rules):
                    continue

                yield from iter_files(entry, current_rules)

            elif entry.is_file():

                # 防止将本次输出文件再次合并进去
                if entry.resolve() == OUTPUT_PATH:
                    continue

                if is_ignored(entry, False, current_rules):
                    continue

                yield entry

        except OSError as exc:
            print(f"[跳过] 无法处理 {entry}: {exc}", file=sys.stderr)


# =========================
# 文件内容读取
# =========================

def read_file(path: Path) -> tuple[str | None, str]:
    """
    读取文件并尽可能识别文本编码。

    支持：
    - UTF-8
    - UTF-8 BOM
    - UTF-16 / UTF-32 BOM
    - GB18030

    二进制文件不读取原始内容。
    """

    raw = path.read_bytes()

    if not raw:
        return "", "utf-8"

    # 优先处理带 BOM 的文本文件
    if raw.startswith(codecs.BOM_UTF32_LE) or \
       raw.startswith(codecs.BOM_UTF32_BE):
        return raw.decode("utf-32"), "utf-32"

    if raw.startswith(codecs.BOM_UTF16_LE) or \
       raw.startswith(codecs.BOM_UTF16_BE):
        return raw.decode("utf-16"), "utf-16"

    if raw.startswith(codecs.BOM_UTF8):
        return raw.decode("utf-8-sig"), "utf-8-sig"

    # 二进制文件通常包含大量 NUL 字节
    if b"\x00" in raw[:8192]:
        return None, "binary"

    # 检查控制字符比例，辅助识别非文本文件
    sample = raw[:8192]

    control_count = sum(
        1 for byte in sample
        if byte < 32 and byte not in (7, 8, 9, 10, 12, 13, 27)
    )

    if control_count / len(sample) > 0.05:
        return None, "binary"

    # 尝试常见文本编码
    for encoding in ("utf-8", "gb18030"):
        try:
            return raw.decode(encoding), encoding
        except UnicodeDecodeError:
            continue

    return None, "unknown"


# =========================
# 合并输出
# =========================

def merge_files():

    file_count = 0
    text_count = 0
    binary_count = 0
    error_count = 0

    print(f"项目目录：{ROOT}")
    print(f"输出文件：{OUTPUT_PATH}")
    print("开始扫描并合并...\n")

    with OUTPUT_PATH.open(
        "w",
        encoding="utf-8",
        newline="\n"
    ) as output:

        for file_path in iter_files(ROOT, []):

            relative_path = file_path.relative_to(ROOT).as_posix()

            file_count += 1

            print(f"[{file_count}] {relative_path}")

            output.write(f"{SEPARATOR}\n")
            output.write(f"{relative_path}\n")
            output.write(f"{SEPARATOR}\n")

            try:
                content, encoding = read_file(file_path)

                if content is None:

                    if encoding == "binary":
                        output.write("[二进制文件：内容已跳过]\n")
                        binary_count += 1
                    else:
                        output.write("[无法识别文件编码：内容已跳过]\n")
                        error_count += 1

                else:
                    output.write(content)

                    # 保证下一个分隔符从新的一行开始
                    if content and not content.endswith("\n"):
                        output.write("\n")

                    text_count += 1

            except Exception as exc:
                output.write(f"[文件读取失败：{exc}]\n")
                error_count += 1

            output.write("\n")

    print("\n========== 合并完成 ==========")
    print(f"扫描文件数：{file_count}")
    print(f"文本文件数：{text_count}")
    print(f"二进制文件：{binary_count}")
    print(f"读取异常数：{error_count}")
    print(f"输出位置：{OUTPUT_PATH}")


if __name__ == "__main__":
    merge_files()

# diary_graph — Yuki 日记概念图谱

外部独立程序组：把 Yuki 的 6000+ 篇日记做成**关键词共现图**，输出一个离线单文件 HTML，
用 vis-network（forceAtlas2Based 物理仿真）呈现"概念星系"，点节点即可看相关日记原文片段。

本分支只做「建图 + 可视化」。话题随机游走 / 自主对抗选目标是后续分支的事。

## 数据来源

直接只读 `../yuki_memory/chroma.sqlite3` 的 `embeddings` / `embedding_metadata` 表
（collection `diaries`），**不依赖 chromadb 包**，也不会加载嵌入模型。

## 安装

```bash
cd diary_graph
python3 -m venv .venv
.venv/bin/pip install -r requirements.txt
```

## 使用

```bash
# 全量建图（默认：全 6485 篇、每篇 8 个关键词、半衰期 365 天、最多 900 节点）
.venv/bin/python -m diary_graph build

# 打开可视化
.venv/bin/python -m diary_graph serve --port 8778
# 浏览器访问 http://127.0.0.1:8779/mind_map.html
```

产物都在 `output/`：

| 文件 | 说明 |
| --- | --- |
| `mind_map.html` | 单文件可视化（内联 vis-network，离线可用），点节点弹出该概念的相关日记 |
| `graph_stats.md` / `.json` | 核心话题、桥梁话题、介数中心性、簇规模 |
| `keywords_cache.json` | 关键词缓存（词典/停用词/别名改动会自动失效） |

## 四阶段流水线

1. **抽取**（`extract.py`）：`jieba` + 自定义词典（`custom_dict.txt`，Yuki 领域词）+ 别名归一（`aliases.txt`）
   + 词性过滤（n/nr/ns/nt/nz/vn/v/eng）+ TF-IDF Top-K + 停用词（`stopwords.txt`，另叠加主项目 `blacklist.txt`）。
2. **建图**（`graph.py`）：同一篇日记的 Top-K 关键词两两连边，无向加权。
3. **优化**：时间衰减 `weight += 2^(-days_ago/半衰期)`；`--min-weight` 剪弱边；清孤岛；`--max-nodes` 按加权度截断。
4. **可视化**（`render.py`）：节点大小 ∝ 度，颜色 = Louvain 簇，边宽/透明度 ∝ 共现权重；
   支持搜索定位、冻结/继续布局、复位视角、点击看日记片段。

## 常用调参

| 参数 | 默认 | 作用 |
| --- | --- | --- |
| `--top-k` | 8 | 每篇取几个关键词，越大图越噪 |
| `--halflife` | 365 | 时间衰减半衰期（天）；`<=0` 关闭衰减 |
| `--min-weight` | 1.0 | 边权重下限，调高图形更干净、簇更碎 |
| `--max-nodes` | 900 | 只保留加权度最高的概念，前端渲染性能相关 |
| `--resolution` | 1.0 | Louvain 分辨率，调大簇更细 |
| `--extra-stop` | `../blacklist.txt` | 额外屏蔽词文件；想屏蔽某些群友名/口头禅就往这里加 |
| `--limit` | 0 | 只用最新 N 篇 |

示例：想看更干净的"话题"而非"人名星系"：

```bash
.venv/bin/python -m diary_graph build --min-weight 3 --max-nodes 800 --resolution 1.3
```

## 已知局限

- jieba 的 `nr` 词性并不可靠（会把"蔚蓝档案""安卓"标成人名），所以没有做自动去人名；
  高频群友名（Momo、白姐姐…）会成为超级枢纽节点，需要就手工加进 `--extra-stop` 文件。
- 全量剪枝后约 2500 节点 / 12000 边，前端一次渲染 1200 节点较流畅，更多会卡。
- 节点相关日记只内嵌片段（默认 260 字、每节点 8 篇）以控制 HTML 体积（约 7MB）。
